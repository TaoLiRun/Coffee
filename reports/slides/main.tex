\section{Data Overview}
\begin{frame}{Features}
\begin{itemize}
    \item \textbf{Time}: 2020/6/1 to 2021/12/31
    \item \textbf{\#stores}: 260
    \item \textbf{Unique members}: 779757
    \begin{itemize}
        \item Demographics: gender, birth year, phone model
        \item Allow push or not
        \item Hierarchical invitation structure (inviter id)
    \end{itemize}
    \item \textbf{Products}
    \begin{itemize}
        \item Coffee, Non-coffee drinks, food
        \item Top commodity or not labelled in the data
        \item New products appearing
    \end{itemize}
    \item \textbf{Orders}: time, member, products, money spent, discounts, \#coupons
    \item \textbf{Pushes}: time, member, type, discount, coupon
\end{itemize}    
\end{frame}

\begin{frame}{Messages from the Company}
\begin{itemize}
    \item New product introduction across stores should be on roughly the same date
    \item Product removal date depends on how fast the product runs out of ingredients at different stores
    \item Different products have different positioning: some are ``base products'', some are ``short-tenure products''
    \item Product lifecycle is pre-determined
    \item Mechanism of push notifications
    \begin{itemize}
        \item A pool of consumers and a budget are first decided
        \item An algorithm is used to determine what kind of push to send to which consumer in the pool, depending on the conversion prediction
    \end{itemize}
\end{itemize}
\end{frame}

\subsection{Stores}
\begin{frame}{Store Information}
\begin{itemize}
    \item There are 260 stores in Nanjing in total, and all of them are in our data
    \item We know
    \begin{itemize}
        \item the exact location of each store (accurate to a building/shopping mall)
        \item the setup time and shutdown time of each store
    \end{itemize}
    \item We know which store a specific order happened
    \begin{itemize}
        \item Data: consumer-time-store-product
    \end{itemize}
    \item Product removal time varies across stores: because of different rates of ingredients consuming
    \item Note: our data only includes a sub-sample of consumers, who entered as a new consumer during this time period, so we can only analyze the causal effect on this sub-group
\end{itemize}
\end{frame}

\begin{frame}{Some Domain Knowledge}
\begin{itemize}
    \item There are two types of stores: self-operation (229) and franchise (31)
    \begin{itemize}
        \item Headquarters make decisions for self-operation stores, e.g. which products to sell
        \item Store heads of  franchise stores also only have limited decision rights on new product introduction
    \end{itemize}
    \item Different stores have different opening and close time, and are differently affected by covid
\end{itemize}    
\end{frame}

\end{itemize}

\section{Weekly Regression}
\begin{frame}{Review}
\begin{itemize}
    \item Direct impact on a store is small and hard to quantify
    \item We can study the causal effect of removing ``top'' products on consumer
    \begin{itemize}
        \item ``top'' is defined in the data
        \item Treatment: consumers whose store removed a ``top'' product
        \item Control: consumers whose store hasn't removed a ``top'' product
    \end{itemize}
    \item Identification: staggered DiD
    \item Potential concerns
    \begin{enumerate}
        \item We infer supply from demand. Should we use the stock-out days also as treatment?
        \item other product changes are happening at the same time
        \item All units are treated in the end, and most units are treated in the last period
    \end{enumerate}
    \item The impact may be small, given that consumers are highly variety-seeking
\end{itemize}    
\end{frame}



\begin{frame}{Takeaway}
\begin{itemize}
    \item Some stores closed on weekends because they were located in office buildings
    \item Stores were affected by covid differently
    \item Product introduction happens simultaneously across stores, while product removal doesn't
    \item Multiple products could be introduced at the same time
    \item Before introduction of new products, product removal always happens
    \begin{itemize}
        \item As documented in Bernard, Redding, and Schott (2010)\footfullcite{bernard2010multiple}
    \end{itemize}
\end{itemize}    
\end{frame}

            \item Fill weeks with no transactions as \texttt{orders = 0}.
            \item Calculate \textbf{Sleep Weeks}: Consecutive weeks with zero orders.
        \end{itemize}
        \item \textbf{Customer Filtering}:
        \begin{itemize}
            \item Remove single-purchase customers (40.68\% of original sample)
            \item Remove customers with $>$14 orders in any week (outliers)
            \item Final sample: 29,054 customers (58.42\% of original)
        \end{itemize}
    \end{itemize}
\end{frame}

\section{Closure Shock}
\subsection{Store Closure Analysis}

\begin{frame}{RQ: Impact of State-dependence on Future purchasing}
\textbf{Motivation}
\begin{itemize}
    \item Natural experiment: store closures during Covid-19
    \item No direct closure data available; need to identify closures from demand patterns
    \item How FORCED disconnection in purchase affects customers' future behavior
    \item And how notifications/new products can make customers reconnect
\end{itemize}

\vspace{0.3cm}

\textbf{Steps}
\begin{itemize}
    \item Identify Covid-forced store closures using demand data as a proxy
    \item Visualise closures geographically to understand spatial patterns
    \item Identify treated customers and the control group using historical purchases
    \item Validate the proxy by examining customer purchase behavior during closures
    \item \textbf{[New]} Quantify the closure impact using statistical tests
\end{itemize}
\end{frame}

\begin{frame}{Methodology (1/2): Identifying Store Closures}
\textbf{Step 1: Geocode Store Locations}
\begin{itemize}
    \item 260 stores in Nanjing with Chinese addresses
    \item Used Baidu Maps API to convert addresses to coordinates
    \item Output: \texttt{nanjing\_stores\_geocoded.csv} with latitude/longitude
    \item 247/260 stores successfully geocoded (95.4\%)
\end{itemize}

\vspace{0.2cm}

\textbf{Step 2: Identify Covid-forced Closures from Demand Data}
\begin{itemize}
    \item Create complete grid: all date $\times$ store combinations
    \item Identify consecutive zero-demand periods $\geq$ 10 days
    \item Filter: stores must have non-zero demand \textbf{before AND after} closure
    \item This excludes stores that opened late or closed permanently
\end{itemize}
\end{frame}

\begin{frame}{Methodology (2/2): Customer Impact Analysis}
\textbf{Step 3: Customer-Store Relationship}
\begin{itemize}
    \item For each customer, identify their most frequently visited store
    \item Calculate ratio of purchases at preferred store
    \item Define ``affected customers'': those with $\geq$ 5 purchases and $\geq$ 80\% at closed store
\end{itemize}

\vspace{0.2cm}

\textbf{Step 4: Impact Validation}
\begin{itemize}
    \item Compare purchase behavior during closure period
    \item \textbf{Affected group}: customers tied to closed store
    \item \textbf{Control group}: who meet purchase threshold but not tied to closed store
    \item Calculate ``selectivity ratio'': affected purchase ratio / control purchase ratio
\end{itemize}
\end{frame}

\begin{frame}{From Closure Detection to Estimation}
\small
\textbf{Estimation sample used in the new reports}
\begin{itemize}
    \item Keep short closures only: duration $<30$ days and closure date after Aug 2020
    \item Match each treated store to up to 5 never-closed control stores by setup time
    \item Treatment: member's preferred store is the closed store
    \item Control: member's preferred store is one of the matched control stores
    \item Event-time panel uses $t \in \{-4,-3,-2,-1,1,2,3,4\}$ and excludes $t=0$ from regressions
\end{itemize}

\vspace{0.2cm}
\textbf{Why this refinement?}
\begin{itemize}
    \item Long closures are much more likely to reflect seasonal shutdowns than short disruption shocks
    \item The matched-control design is what feeds the DiD and DDD specifications in the four updated reports
\end{itemize}
\end{frame}

\begin{frame}{Additional Design Updates}
\small
\textbf{What changed relative to the earlier closure analysis}
\begin{itemize}
    \item Compare multiple horizons ($h=1,2,4$) instead of relying on one pre/post window
    \item Use XGBoost to classify members likely to purchase during the closure counterfactual
    \item Pre-period validation accuracy is about 73\% in the purchase-displacement model
    \item Heterogeneity checks by closure length and push exposure do not reveal stronger habit-disruption effects
\end{itemize}

\vspace{0.2cm}
\textbf{Key caveat absorbed into the new design}
\begin{itemize}
    \item Many long university-store closures are holiday-driven, so the current estimation focuses on short closures
    \item This is why the updated empirical section emphasizes the 22 retained short closures rather than all 101 raw shutdown spells
\end{itemize}
\end{frame}
